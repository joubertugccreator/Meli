Usage Example

// Import the AIAvatar class
import { AIAvatar } from './src/AIAvatar.js';

// Create an avatar instance
const avatar = new AIAvatar('#container', {
  size: 200,
  primaryColor: '#6366f1',
  showParticles: true
});

// Change states
avatar.think();  // Processing state
avatar.speak();  // Speaking state
avatar.listen(); // Listening state
avatar.idle();   // Back to idle
